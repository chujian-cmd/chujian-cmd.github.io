  function makeFrame(cfg, isSingle) {
    var frame = document.createElement('div');
    frame.className = (cfg && cfg.mode === 'buttons') ? 'frame button' : (isSingle ? 'frame single' : 'frame');
    var el = document.createElement('liquid-glass');
    if (window.LG_WALLPAPER) el.setAttribute('wallpaper', window.LG_WALLPAPER);
    el.setAttribute('mode', cfg.mode);
    frame.appendChild(el);
    if (cfg.tabs) requestAnimationFrame(function() { el.setTabs(cfg.tabs); });
    if (cfg.buttons) requestAnimationFrame(function() { el.setButtons(cfg.buttons); });
    if (cfg.dialog) requestAnimationFrame(function() { el.setDialog(cfg.dialog); });
    return frame;
  }

  var _loadTimer = 0;
  function showLoading() {
    stage.innerHTML = '<div class="loading"><div class="spinner"></div>加载中...</div>';
  }

  function select(group) {
    nav.querySelectorAll('button').forEach(function (b) {
      b.classList.toggle('active', b.getAttribute('data-group') === group);
    });
    showLoading();
    clearTimeout(_loadTimer);
    _loadTimer = setTimeout(function() {
      var g = GROUPS[group];
      var oldEls = stage.querySelectorAll('liquid-glass');
      for (var o = 0; o < oldEls.length; o++) oldEls[o].remove();
      stage.innerHTML = '';
      if (g.pairs) {
        var pair = document.createElement('div');
        pair.className = 'pair';
        g.pairs.forEach(function(cfg) {
          var item = document.createElement('div');
          item.className = 'pair-item';
          var label = document.createElement('div');
          label.className = 'label';
          label.textContent = cfg.tag;
          item.appendChild(label);
          item.appendChild(makeFrame(cfg, false));
          pair.appendChild(item);
        });
        stage.appendChild(pair);
      } else if (g.buttons) {
        var list = document.createElement('div');
        list.className = 'pair';
        g.buttons.forEach(function(b) {
          var item = document.createElement('div');
          item.className = 'pair-item';
          var label = document.createElement('div');
          label.className = 'label';
          label.textContent = b.label;
          item.appendChild(label);
          item.appendChild(makeFrame({ mode: 'buttons', buttons: [b] }, true));
          list.appendChild(item);
        });
        stage.appendChild(list);
      } else if (g.single) {
        stage.appendChild(makeFrame(g.single, true));
      }
    }, 50);
  }

  nav.addEventListener('click', function (e) {
    var btn = e.target.closest('button');
    if (btn) select(btn.getAttribute('data-group'));
  });

  select('toggle');

  var _activeGroup = 'toggle';
  var _origSelect = select;
  select = function(g) { _activeGroup = g; _origSelect(g); };
  new MutationObserver(function(mutations) {
    if (_activeGroup !== 'dialog') return;
    mutations.forEach(function(m) {
      if (m.type === 'attributes' && m.attributeName === 'mode') {
        var el = m.target;
        if (el.getAttribute('mode') !== 'dialog') {
          requestAnimationFrame(function() { el.setAttribute('mode', 'dialog'); });
        }
      }
    });
  }).observe(document.body, { attributes: true, subtree: true, attributeFilter: ['mode'] });
